cmake_minimum_required(VERSION 3.20)
project(veyra VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(VEYRA_BUILD_BENCH "Build benchmark harness" ON)
set(VEYRA_PROFILE "desktop" CACHE STRING "desktop, minimal, embedded, or headless")
set_property(CACHE VEYRA_PROFILE PROPERTY STRINGS desktop minimal embedded headless)
set(_veyra_profiles desktop minimal embedded headless)
if(NOT VEYRA_PROFILE IN_LIST _veyra_profiles)
 message(FATAL_ERROR "Invalid VEYRA_PROFILE: ${VEYRA_PROFILE}")
endif()

include("${CMAKE_CURRENT_SOURCE_DIR}/cmake/deps.cmake")

# A desktop profile prefers a real GPU, but never requires one to configure:
# without SDL3 it falls back to software and says so.
if(VEYRA_HERMETIC)
 set(SDL3_FOUND TRUE)
else()
 find_package(SDL3 3.2 CONFIG QUIET COMPONENTS SDL3)
endif()
set(_veyra_desktop_default OFF)
if(VEYRA_PROFILE STREQUAL "desktop")
 set(_veyra_desktop_default ON)
endif()
set(_veyra_sdl3_default OFF)
set(_veyra_gpu_default OFF)
if(_veyra_desktop_default AND SDL3_FOUND)
 set(_veyra_sdl3_default ON)
 set(_veyra_gpu_default ON)
elseif(VEYRA_PROFILE STREQUAL "desktop")
 message(STATUS "Veyra: SDL3 3.2+ not found; the desktop profile builds software rendering only")
endif()

option(VEYRA_WITH_FILESYSTEM "Build asynchronous filesystem enumeration support" OFF)
option(VEYRA_WITH_TEXT "Build HarfBuzz text shaping support" OFF)
option(VEYRA_WITH_UNICODE "Build ICU-based Unicode editing, bidi, and line breaking" OFF)
option(VEYRA_WITH_FREETYPE "Build FreeType glyph rasterization support" OFF)
option(VEYRA_WITH_SDL3 "Build the SDL3 window backend" ${_veyra_sdl3_default})
option(VEYRA_WITH_GLFW "Build the GLFW platform window backend" OFF)
option(VEYRA_GPU_SDL "Build the SDL_GPU display-list consumer" ${_veyra_gpu_default})
option(VEYRA_WITH_WEBGPU "Build the WebGPU display-list consumer" OFF)
option(VEYRA_WITH_CODECS "Build vendored image decoding support" ON)
option(VEYRA_WITH_SVG "Build vendored nanosvg SVG rasterization support" ON)
set(VEYRA_LIBWEBP_URL "https://storage.googleapis.com/downloads.webmproject.org/releases/webp/libwebp-1.6.0.tar.gz" CACHE STRING "libwebp pinned tarball")
set(VEYRA_LIBWEBP_SHA256 "e4ab7009bf0629fd11982d4c2aa83964cf244cffba7347ecd39019a9e38c4564" CACHE STRING "libwebp pinned tarball sha256")
option(VEYRA_WITH_WEBP "Build WebP image support (fetched if missing)" ON)
option(VEYRA_WITH_AVIF "Build AVIF image support (fetched if missing)" ON)
set(VEYRA_LIBAVIF_URL "https://github.com/AOMediaCodec/libavif/archive/refs/tags/v1.4.2.tar.gz" CACHE STRING "libavif pinned tarball")
set(VEYRA_LIBAVIF_SHA256 "2b645287340ba5a631d268b551dc2d72bd73ac33335962dd36dcdb6d8366921d" CACHE STRING "libavif pinned tarball sha256")
if(VEYRA_GPU_SDL)
 set(VEYRA_WITH_SDL3 ON)
endif()

if(VEYRA_WITH_SDL3)
 veyra_dependency_sdl3()
 if(NOT TARGET SDL3::SDL3)
  message(FATAL_ERROR "VEYRA_WITH_SDL3=ON but SDL3::SDL3 is unavailable")
 endif()
endif()

enable_testing()

set(veyra_build_config "$<CONFIG>")
if(NOT CMAKE_CONFIGURATION_TYPES AND NOT CMAKE_BUILD_TYPE)
 set(veyra_build_config "unspecified")
endif()

find_package(Threads REQUIRED)

include("${CMAKE_CURRENT_SOURCE_DIR}/cmake/targets/helpers.cmake")
include("${CMAKE_CURRENT_SOURCE_DIR}/cmake/targets/core.cmake")
include("${CMAKE_CURRENT_SOURCE_DIR}/cmake/targets/backends.cmake")
include("${CMAKE_CURRENT_SOURCE_DIR}/cmake/targets/text.cmake")
include("${CMAKE_CURRENT_SOURCE_DIR}/cmake/targets/codecs.cmake")
include("${CMAKE_CURRENT_SOURCE_DIR}/cmake/targets/webgpu.cmake")
include("${CMAKE_CURRENT_SOURCE_DIR}/cmake/targets/tests.cmake")
