cmake_minimum_required(VERSION 3.20)
project(veyra VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(VEYRA_BUILD_BENCH "Build benchmark harness" ON)
option(VEYRA_WITH_FILESYSTEM "Build asynchronous filesystem enumeration support" OFF)
option(VEYRA_WITH_TEXT "Build HarfBuzz text shaping support" OFF)
option(VEYRA_WITH_UNICODE "Build ICU-based Unicode editing, bidi, and line breaking" OFF)
option(VEYRA_WITH_FREETYPE "Build FreeType glyph rasterization support" OFF)
option(VEYRA_WITH_SDL3 "Build the SDL3 software window backend" OFF)
option(VEYRA_WITH_GLFW "Build the GLFW platform window backend" OFF)
option(VEYRA_WITH_VULKAN "Build the Vulkan offscreen rect prototype" OFF)
option(VEYRA_WITH_CODECS "Build vendored image decoding support" ON)
option(VEYRA_WITH_SVG "Build vendored nanosvg SVG rasterization support" ON)
set(VEYRA_LIBWEBP_URL "https://storage.googleapis.com/downloads.webmproject.org/releases/webp/libwebp-1.6.0.tar.gz" CACHE STRING "libwebp pinned tarball")
set(VEYRA_LIBWEBP_SHA256 "e4ab7009bf0629fd11982d4c0aa83964cf244cffba7347ecd39019a9e38c456" CACHE STRING "libwebp pinned tarball sha256")
option(VEYRA_WITH_WEBP "Build WebP image support (fetched if missing)" ON)
option(VEYRA_WITH_AVIF "Build AVIF image support (fetched if missing)" ON)
set(VEYRA_LIBAVIF_URL "https://github.com/AOMediaCodec/libavif/archive/refs/tags/v1.4.2.tar.gz" CACHE STRING "libavif pinned tarball")
set(VEYRA_LIBAVIF_SHA256 "2b645287340ba5a631d0b55508b1d2f7bd73ac6b6d8366d0be6921d" CACHE STRING "libavif pinned tarball sha256")
set(VEYRA_PARADIGM "arena" CACHE STRING "arena, reactive, ecs, or all")
set_property(CACHE VEYRA_PARADIGM PROPERTY STRINGS arena reactive ecs all)
set(valid_paradigms arena reactive ecs all)

if(NOT VEYRA_PARADIGM IN_LIST valid_paradigms)
 message(FATAL_ERROR "Invalid VEYRA_PARADIGM: ${VEYRA_PARADIGM}")
endif()

enable_testing()

set(veyra_build_config "$<CONFIG>")
if(NOT CMAKE_CONFIGURATION_TYPES AND NOT CMAKE_BUILD_TYPE)
 set(veyra_build_config "unspecified")
endif()

find_package(Threads REQUIRED)

include("${CMAKE_CURRENT_SOURCE_DIR}/cmake/targets/helpers.cmake")
include("${CMAKE_CURRENT_SOURCE_DIR}/cmake/targets/core.cmake")
include("${CMAKE_CURRENT_SOURCE_DIR}/cmake/targets/backends.cmake")
include("${CMAKE_CURRENT_SOURCE_DIR}/cmake/targets/text.cmake")
include("${CMAKE_CURRENT_SOURCE_DIR}/cmake/targets/codecs.cmake")
include("${CMAKE_CURRENT_SOURCE_DIR}/cmake/targets/vulkan.cmake")
include("${CMAKE_CURRENT_SOURCE_DIR}/cmake/targets/tests.cmake")
